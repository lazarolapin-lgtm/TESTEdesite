function card(icon,title,text){

  return `

      <div class="thumb">

      </div>

          ${title}

/* =========================
   TROCAR PÁGINA
========================= */

function go(p){

  document
    .querySelectorAll('nav button')
    .forEach(

      b => b.classList.toggle(
        'active',
        b.dataset.page === p
      )

    );

  document
    .getElementById('content')
    .innerHTML = pages[p];

  window.scrollTo({
    top:0,
    behavior:'smooth'
  });

}


/* =========================
   MENU
========================= */

document
  .querySelectorAll('nav button')
  .forEach(

    b => {

      b.onclick = () =>
        go(b.dataset.page);

    }

  );


/* =========================
